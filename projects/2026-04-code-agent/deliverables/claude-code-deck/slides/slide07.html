def agent_loop(query):

messages = [{"role":"user",

"content": query}]

while True:

resp = client.messages.create(

model=MODEL,

tools=TOOLS,

messages=messages,

)

messages.append(resp.content)

if resp.stop_reason != "tool_use":

return # done

results = run_tools(resp.content)

messages.append(results)
