# Dispatch map

TOOL_HANDLERS = {

"bash": run_bash,

"read": read_file,

"write": write_file,

"edit": edit_file,

}

def run_tools(blocks):

for b in blocks:

fn = TOOL_HANDLERS[b.name]

yield fn(b.input)
